# 核心功能详细设计文档

## 一、出厂查询功能详细设计

### 1.1 功能概述

出厂查询功能通过设备的IMEI号或序列号(SN)，查询设备的出厂信息，包括：
- 设备基本信息（品牌、型号、颜色、容量）
- 生产信息（生产日期、产地、工厂）
- 保修信息（保修状态、到期时间、剩余天数）
- 激活信息（首次激活日期、激活状态）

### 1.2 查询方式

#### 方式一：本机自动查询
```
用户点击"查询本机" → 自动获取本机IMEI/SN → 调用查询接口 → 展示结果
```

#### 方式二：手动输入查询
```
用户输入IMEI或SN → 选择品牌类型 → 调用查询接口 → 展示结果
```

#### 方式三：扫码查询
```
用户扫描设备包装盒/设置页面的二维码 → 解析IMEI/SN → 调用查询接口 → 展示结果
```

```typescript
    // 3. 调用查询接口

    showResult(response.data);

    // 错误处理
    showError(error.message);
  } finally {
    hideLoading();
  }
}
```

### 1.7 验证规则

#### IMEI验证
```typescript
function validateIMEI(imei: string): boolean {
  // 长度验证
  if (imei.length !== 15) return false;

  // 数字验证
  if (!/^\d{15}$/.test(imei)) return false;

  // Luhn算法校验
  let sum = 0;
  for (let i = 0; i < 15; i++) {
    let digit = parseInt(imei[i]);
    if (i % 2 === 1) {
      digit *= 2;
      if (digit > 9) digit -= 9;
    }
    sum += digit;
  }
  return sum % 10 === 0;
}
```

#### 序列号验证
```typescript
function validateSN(sn: string, brand: string): boolean {
  const patterns = {
    apple: /^[A-Z0-9]{8,12}$/,      // Apple: 8-12位字母数字
    huawei: /^[A-Z0-9]{16,20}$/,    // 华为: 16-20位
    xiaomi: /^[A-Z0-9]{10,20}$/,    // 小米: 10-20位
    samsung: /^[A-Z0-9]{11,15}$/,   // 三星: 11-15位
  };
  return patterns[brand]?.test(sn) ?? /^[A-Z0-9]{8,20}$/.test(sn);
}
```

## 二、电池健康分析功能详细设计

### 2.1 功能概述

电池健康分析功能实时监测设备电池状态，分析电池健康程度，提供：
- 实时电池状态（电量、温度、电压、电流）
- 电池健康度评估（健康百分比、损耗程度）
- 电池容量信息（设计容量、实际容量、损耗容量）
- 充电循环次数
- 健康趋势分析
- 异常预警提醒

```kotlin
        val actualCapacity = getActualCapacity()
        val cycleCount = getCycleCount()

            level = level,
            voltage = voltage,
            temperature = temperature,
            current = current,
            isCharging = isCharging,
            health = health,
            healthPercent = healthPercent,
            designCapacity = designCapacity,
            actualCapacity = actualCapacity,
            cycleCount = cycleCount
        )

    private fun getTemperature(): Float {
        val intent = context.registerReceiver(null, IntentFilter(Intent.ACTION_BATTERY_CHANGED))
        val temp = intent?.getIntExtra(BatteryManager.EXTRA_TEMPERATURE, 0) ?: 0
        return temp / 10f  // 转换为摄氏度

    private fun getDesignCapacity(): Int {
        // 从系统文件读取设计容量
        val file = File("/sys/class/power_supply/battery/charge_full_design")
        if (file.exists()) {
            return file.readText().trim().toInt() / 1000  // 转换为mAh
        }
        return getDefaultCapacity()  // 返回默认值
    }

    private fun getCycleCount(): Int {
        // 尝试从不同路径读取循环次数
        val paths = listOf(
            "/sys/class/power_supply/battery/cycle_count",
            "/sys/class/power_supply/battery/battery_cycle_count"
        )
        for (path in paths) {
            val file = File(path)
            if (file.exists()) {
                return file.readText().trim().toInt()
```

### 2.5 健康度计算算法

```typescript
/**
 * 电池健康度计算
 *
 * 方法一：容量比法（推荐）
 * 健康度 = (实际容量 / 设计容量) × 100%
 *
 * 方法二：循环次数估算法（备用）
 * 健康度 = 100% - (循环次数 / 最大循环次数) × 损耗系数
 */
function calculateBatteryHealth(params: {
  designCapacity: number;
  actualCapacity: number;
  cycleCount?: number;
}): number {
  const { designCapacity, actualCapacity, cycleCount } = params;

  // 方法一：容量比法
  const healthByCapacity = (actualCapacity / designCapacity) * 100;

  // 方法二：循环次数估算法（如果有循环次数数据）
  if (cycleCount && cycleCount > 0) {
    const maxCycle = 500;  // 假设最大循环次数
    const degradationRate = 0.2;  // 每循环损耗率
    const healthByCycle = 100 - (cycleCount / maxCycle * 100 * degradationRate);

    // 取两种方法的平均值
    return Math.round((healthByCapacity + healthByCycle) / 2);
  }

  return Math.round(healthByCapacity);
}
```

```typescript
      temperature: info.temperature,
      voltage: info.voltage,
      healthPercent: info.healthPercent,
      cycleCount: info.cycleCount,
    };

  }
}
```

### 2.7 预警机制

#### 预警规则
```typescript
const ALERT_RULES = {
  // 温度过高预警
  temperature: {
    warning: { threshold: 40, message: '电池温度偏高，建议暂停使用' },
    danger: { threshold: 45, message: '电池温度过高，请立即停止使用' },
  },

  // 健康度预警
  health: {
    warning: { threshold: 80, message: '电池健康度下降，建议关注' },
    danger: { threshold: 60, message: '电池健康度严重下降，建议更换' },
  },

  // 循环次数预警
  cycle: {
    warning: { threshold: 400, message: '充电循环次数较多' },
    danger: { threshold: 500, message: '充电循环次数过多，建议更换电池' },
  },

  // 充电时间过长预警
  charging: {
    warning: { threshold: 240, message: '充电时间过长，请检查充电器' },
  },
};

function checkAlerts(info: BatteryInfo): Alert[] {
  const alerts: Alert[] = [];

  // 温度检查
  if (info.temperature >= ALERT_RULES.temperature.danger.threshold) {
    alerts.push({ level: 'danger', type: 'temperature', message: ALERT_RULES.temperature.danger.message });
  } else if (info.temperature >= ALERT_RULES.temperature.warning.threshold) {
    alerts.push({ level: 'warning', type: 'temperature', message: ALERT_RULES.temperature.warning.message });
  }

  // 健康度检查
  if (info.healthPercent <= ALERT_RULES.health.danger.threshold) {
    alerts.push({ level: 'danger', type: 'health', message: ALERT_RULES.health.danger.message });
  } else if (info.healthPercent <= ALERT_RULES.health.warning.threshold) {
    alerts.push({ level: 'warning', type: 'health', message: ALERT_RULES.health.warning.message });
  }

  return alerts;
}
```

```typescript
async function handleQuery(params: QueryParams) {
  try {
    const result = await queryDevice(params);
    return { success: true, data: result };
  } catch (error) {
    const errorCode = identifyError(error);
    const errorMessage = ERROR_CODES[errorCode].message;

    // 记录错误日志
    logError(error);

    // 显示用户提示
    showToast(errorMessage);

    return { success: false, error: errorCode, message: errorMessage };
  }
}
```
